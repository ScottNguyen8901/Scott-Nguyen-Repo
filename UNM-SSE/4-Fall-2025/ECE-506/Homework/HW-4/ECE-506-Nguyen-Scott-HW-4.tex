	\textbf{Problem 1. Stepsizes and convergence rates for line search methods}
	
	\begin{enumerate}[label=\textbf{1(\alph*)}]
		\item Consider the general equation form
		\[
		f(\bm{x}) = a x_1^2 + b x_2^2 + c x_1 + d x_2 + e.
		\]
		Compute the optimal point $\bm{x}^\ast$ and provide conditions that guarantee optimality.
		\textit{Solution.}
		\[
		\frac{\partial f}{\partial x_1}=2a x_1 + c,\qquad
		\frac{\partial f}{\partial x_2}=2b x_2 + d,\qquad
		\nabla^2 f=\begin{bmatrix}2a&0\\[2pt]0&2b\end{bmatrix}.
		\]
		Setting the gradient to zero gives
		\[
		x_1^\ast=-\frac{c}{2a},\qquad x_2^\ast=-\frac{d}{2b}.
		\]
		This stationary point is a (unique) minimizer when \(a>0\) and \(b>0\) (Hessian \(\succ 0\)); 
		it is a maximizer when \(a<0\) and \(b<0\); otherwise it is a saddle or unbounded in at least one direction.
		
		\item Compute the optimal stepsize for steepest descent line search.
		\textit{Solution.}
		Let $g_k=\nabla f(x_k)$, $H=\nabla^2 f$. For steepest descent with exact line search and $p_k=-g_k$,
		\[
		\boxed{\;\alpha_k^\ast=\frac{g_k^\top g_k}{g_k^\top H g_k}\;}
		\]
		so for $f(x)=a x_1^2+b x_2^2+c x_1+d x_2+e$ with $g_k=\begin{bmatrix}2a x_{1,k}+c\\[2pt]2b x_{2,k}+d\end{bmatrix}$ and
		$H=\begin{bmatrix}2a&0\\[2pt]0&2b\end{bmatrix}$,
		\[
		\boxed{\;\alpha_k^\ast=
			\frac{(2a x_{1,k}+c)^2+(2b x_{2,k}+d)^2}{\,2a(2a x_{1,k}+c)^2+2b(2b x_{2,k}+d)^2\,}\;}
		\]
		(positive when $a>0,\;b>0$).

		\item For our quadratic case, provide:
		\begin{enumerate}[label=\roman*.]
			\item An expression for $\|\cdot\|_Q$.\\
			\textit{Solution}
			\[
			Q=\begin{bmatrix}a&0\\[2pt]0&b\end{bmatrix},\qquad
			\|z\|_Q \;=\; \sqrt{z^\top Q z}
			\;=\; \sqrt{a\,z_1^2 + b\,z_2^2}.
			\]
			\item An expression in terms of distance from $\bm{x}^\ast$.\\
			\textit{Solution}
			\[
			f(x)-f(x^\ast)
			=\,(x-x^\ast)^\top Q\,(x-x^\ast)
			=\;a\,(x_1-x_1^\ast)^2 + b\,(x_2-x_2^\ast)^2 .
			\]
			
			\item The best convergence case in terms of $a,b$.\\
			\textit{Solution}
			The best case occurs when the quadratic is isotropic, i.e.\ \(a=b>0\), which gives a condition number of 1 and the fastest convergence.
			
			\item The worst-conditioned case in terms of $a,b$, assuming $a>b$.\\
			\textit{Solution}
			The worst case occurs as the ratio \(a/b\) becomes large (with \(a>b>0\)), leading to a high condition number and slow convergence.

		\item Repeat 1(d) for Newton's method. Do you need two steps? Explain using the Newton convergence theorem.\\
		\textit{Solution}
		For \(f(x)=x_1^2+10x_2^2\) we have
		\[
		g(x)=\nabla f(x)=\begin{bmatrix}2x_1\\20x_2\end{bmatrix},\qquad
		H=\nabla^2 f(x)=\begin{bmatrix}2&0\\0&20\end{bmatrix},\quad
		H^{-1}=\begin{bmatrix}\tfrac12&0\\[2pt]0&\tfrac1{20}\end{bmatrix}.
		\]
		Newton’s update is
		\[
		x_{k+1}=x_k - H^{-1}g(x_k)
		= x_k - \begin{bmatrix}\tfrac12&0\\[2pt]0&\tfrac1{20}\end{bmatrix}
		\begin{bmatrix}2x_{k,1}\\[2pt]20x_{k,2}\end{bmatrix}
		= x_k - \begin{bmatrix}x_{k,1}\\ x_{k,2}\end{bmatrix}
		= \begin{bmatrix}0\\0\end{bmatrix}.
		\]
		Thus, starting from \(x_0=x^\ast+[1,1]^T=[1,1]^T\), Newton gives \(x_1=\mathbf{0}=x^\ast\) in \emph{one} step.
		
		For a strictly convex quadratic, the Hessian is constant and positive definite; the Newton step solves \(\nabla f(x)=0\) exactly in one iteration. More generally with a start sufficiently close to \(x^\ast\), Newton converges quadratically; the constant-Hessian case collapses to one-step convergence.
		
	\end{enumerate}

	Across random initializations, the three methods exhibit the expected trade–offs in robustness, efficiency, and accuracy. On convex, well–conditioned problems (e.g., \emph{Sphere}, \emph{Booth}), BFGS and Newton converge in a few steps with superlinear to quadratic rates, while Steepest Descent (SD) converges reliably but requires tens of iterations. On harder nonconvex landscapes (e.g., \emph{Beale}, \emph{Goldstein–Price}), SD is the most forgiving yet typically only linear and slow; BFGS can stall without a good basin or strong–Wolfe safeguards; and Newton succeeds only when the Hessian is informative/positive–definite (otherwise it fails or hits iteration caps). Efficiency reflects this: SD uses \(O(n)\) memory (about \(3n\) floats) with no Hessian cost but pays in iteration count; BFGS and Newton require \(O(n^2)\) memory (roughly \(n^2{+}3n\) and \(n^2{+}2n\), respectively) and higher per–step work (BFGS via frequent function/gradient evaluations; Newton via one Hessian per step) that is justified only when they are within the attraction region. Accuracy results align with theory when they are: SD is linear, BFGS is superlinear, and Newton is quadratic near the solution; applying both error–based and gradient–based rate estimates yields consistent qualitative conclusions, with discrepancies mainly before the asymptotic regime or in flat/ill–conditioned regions.

function H = num_hess(gradfun, x)
h   = 1e-6;
H   = zeros(2,2);
e1  = [1; 0];
e2  = [0; 1];
g1p = gradfun(x + h*e1);
g1m = gradfun(x - h*e1);
g2p = gradfun(x + h*e2);
g2m = gradfun(x - h*e2);
H(:,1) = (g1p - g1m) / (2*h);
H(:,2) = (g2p - g2m) / (2*h);
H = (H + H.') / 2;
end

function [xk, Xall, Dhist, counts] = tr_cg_driver(fun, grad, hess, x0, tol, TR)
xk    = x0(:);
Xall  = xk;
D     = TR.Delta0;
Dhist = D;
counts = struct('f',0,'g',0,'H',0);
for k = 1:TR.maxit
gk = grad(xk); counts.g = counts.g + 1;
if norm(gk) <= tol, break; end
Bk = hess(xk); counts.H = counts.H + 1;
[p, ~]  = CG_Steihaug(TR.cg_eps, gk, Bk, D);
mk0 = 0; mkp = gk.'*p + 0.5*p.'*Bk*p;
denom = -(mkp - mk0); if denom <= 0, denom = eps; end
fx = fun(xk); counts.f = counts.f + 1;
fnew = fun(xk + p); counts.f = counts.f + 1;
rho = (fx - fnew) / denom;
accepted = false;
if rho >= TR.eta1
xk = xk + p; Xall(:, end+1) = xk; %#ok<AGROW>
accepted = true;
else
D = TR.gamma1 * D;
end
if rho < 0.25
D = max(TR.gamma1 * D, 1e-12);
elseif (rho > TR.eta2) && (abs(norm(p) - D) <= 1e-12)
D = min(TR.gamma2 * D, TR.DeltaMax);
end
Dhist(end+1) = D; %#ok<AGROW>
if accepted
gk2 = grad(xk); counts.g = counts.g + 1;
if norm(gk2) <= tol, break; end
end
end
end
